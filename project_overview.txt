@st.cache_resource
def get_predictor_instance(arch_name: str, threshold: float):
    """

if WEBRTC_AVAILABLE:
    class LiveFoodVideoProcessor(VideoProcessorBase):
        """Processes continuous live frames from browser webcam using shared predictor."""

        def __init__(self):
            self.predictor = None
            self.last_result = None
            self.frame_count = 0

        def set_predictor(self, predictor: FoodPredictor):
            self.predictor = predictor

        def recv(self, frame: av.VideoFrame) -> av.VideoFrame:
            img = frame.to_ndarray(format="bgr24")
            self.frame_count += 1
            h, w = img.shape[:2]

            # Center-crop coordinates
            box_size = int(min(h, w) * 0.65)
            x1 = (w - box_size) // 2
            y1 = (h - box_size) // 2
            x2 = x1 + box_size
            y2 = y1 + box_size

            # Predict every 3rd frame
            if self.predictor is not None and (self.frame_count % 3 == 0 or self.last_result is None):
                crop = img[y1:y2, x1:x2]
                self.last_result = self.predictor.predict_cv2(crop, smooth=True)

            # Draw visual HUD on frame
            if self.last_result is not None:
                res = self.last_result
                if res["is_food"]:
                    color = (0, 220, 0)  # Green
                    text = f"{res['class_name']} ({res['confidence']*100:.1f}%)"
                    sub_text = f"Rs. {res['price']}  |  {res['kcal']} kcal"
                else:
                    color = (0, 0, 230)  # Red
                    text = "Unknown food"
                    sub_text = "No food / Unknown food"

                # Draw targeting box
                cv2.rectangle(img, (x1, y1), (x2, y2), color, 2)
                cv2.putText(
                    img, text, (x1, max(30, y1 - 10)),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.7, color, 2, cv2.LINE_AA
                )
                cv2.putText(
                    img, sub_text, (x1, y2 + 25),
                    cv2.FONT_HERSHEY_SIMPLEX, 0.55, (230, 230, 230), 1, cv2.LINE_AA
                )

            return av.VideoFrame.from_ndarray(img, format="bgr24")

def add_item_to_bill(dish_class: str, display_name: str, price: int, kcal: int):
    """
    Adds or increments an item in the running bill.
    Safety rule: 'not_food' or price 0 items are NEVER added.
    """
    if dish_class == "not_food" or price <= 0:
        return

    for item in st.session_state.bill_items:
        if item["class"] == dish_class:
            item["qty"] += 1
            item["total_price"] = item["qty"] * item["price"]
            item["total_kcal"] = item["qty"] * item["kcal"]
            return

    st.session_state.bill_items.append({
        "class": dish_class,
        "name": display_name,
        "qty": 1,
        "price": price,
        "kcal": kcal,
        "total_price": price,
        "total_kcal": kcal
    })


def clear_bill():
    """Resets the running bill."""
    st.session_state.bill_items = []

def parse_args():

    parser.add_argument(
        "--arch",
        type=str,
        default="mobilenetv3",
        choices=["mobilenetv3", "efficientnet_b0"],
        help="Model architecture for PyTorch backend. Default: mobilenetv3",

    parser.add_argument(

    parser.add_argument(
        "--threshold",

    parser.add_argument(
        "--device",
        type=str,

    )

def draw_hud_banner(frame, x, y, w, h, bg_color=(20, 20, 20), alpha=0.65):
    """Draws a semi-transparent dark banner behind text for high readability."""
    import cv2
    sub_img = frame[y:y + h, x:x + w]
    rect = cv2.rectangle(
        sub_img.copy(), (0, 0), (w, h), bg_color, -1
    )
    frame[y:y + h, x:x + w] = cv2.addWeighted(rect, alpha, sub_img, 1 - alpha, 0)


def main():

    print("=" * 65)

    print(f"Backend Engine   : {args.backend.upper()}")
    print(f"Architecture     : {args.arch}")
    print(f"Safety Threshold : {args.threshold * 100:.0f}%")
    print(f"Camera Index     : {args.camera}")

    # Check for PyTorch CUDA device

# PYTORCH MODEL ARCHITECTURE BUILDER
# ==============================================================================
def build_pytorch_architecture(arch_name: str, num_classes: int):
    """
    Constructs MobileNetV3-Large or EfficientNet-B0 matching Colab training.
    Replaces classifier layer to output num_classes (default 16).
    """
    try:
        import torch.nn as nn
        from torchvision import models
    except ImportError as e:
        raise ImportError(
            "PyTorch or torchvision is not installed. For NVIDIA RTX 5050 (CUDA 12.8), run:\n"
            "pip install torch torchvision --index-url https://download.pytorch.org/whl/cu128"
        ) from e

    if arch_name == "mobilenetv3":
        model = models.mobilenet_v3_large(weights=None)
        in_features = model.classifier[3].in_features
        model.classifier[3] = nn.Linear(in_features, num_classes)
        target_layer = model.features[-1]
    elif arch_name == "efficientnet_b0":
        model = models.efficientnet_b0(weights=None)
        in_features = model.classifier[1].in_features
        model.classifier[1] = nn.Linear(in_features, num_classes)
        target_layer = model.features[-1]
    else:
        raise ValueError(f"Unsupported architecture '{arch_name}'. Use 'mobilenetv3' or 'efficientnet_b0'.")

    return model, target_layer


# ==============================================================================
# PREPROCESSING & NUMPY SOFTMAX
# ==============================================================================
def center_crop_square(img_bgr_or_rgb: np.ndarray) -> np.ndarray:
    """Center crops an image array (height, width, channels) into a 1:1 square."""
    h, w = img_bgr_or_rgb.shape[:2]
    min_dim = min(h, w)
    start_x = (w - min_dim) // 2
    start_y = (h - min_dim) // 2
    return img_bgr_or_rgb[start_y:start_y + min_dim, start_x:start_x + min_dim]


def preprocess_numpy_image(img_rgb: np.ndarray, target_size: Tuple[int, int] = (224, 224)) -> np.ndarray:
    """
    Center crops, resizes to target_size, scales to [0, 1], applies ImageNet normalization,
    and returns a float32 array with shape [1, 3, 224, 224].
    """
    import cv2

    square = center_crop_square(img_rgb)
    resized = cv2.resize(square, target_size, interpolation=cv2.INTER_LINEAR)
    # Scale to [0, 1]
    norm = resized.astype(np.float32) / 255.0
    # Normalize with ImageNet mean and std
    norm = (norm - IMAGENET_MEAN) / IMAGENET_STD
    # HWC to CHW -> [3, 224, 224]
    chw = np.transpose(norm, (2, 0, 1))
    # Add batch dimension -> [1, 3, 224, 224]
    return np.expand_dims(chw, axis=0)


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax for 1D or 2D numpy arrays."""
    e_x = np.exp(x - np.max(x, axis=-1, keepdims=True))
    return e_x / np.sum(e_x, axis=-1, keepdims=True)


# ==============================================================================
# GRAD-CAM EXPLAINABILITY
# ==============================================================================
def compute_gradcam(model, target_layer, image_pil: Image.Image) -> Tuple[Optional[np.ndarray], Optional[str]]:
    """
    Computes Grad-CAM heatmap overlay for PyTorch models.
    Returns (heatmap_rgb_uint8, error_message).
    """
    try:
        import torch
        from pytorch_grad_cam import GradCAM
        from pytorch_grad_cam.utils.image import show_cam_on_image

        rgb_pil = image_pil.convert("RGB").resize((224, 224))
        rgb_float = np.float32(rgb_pil) / 255.0

        # Build tensor
        tensor_img = (rgb_float - IMAGENET_MEAN) / IMAGENET_STD
        tensor_img = np.transpose(tensor_img, (2, 0, 1))
        tensor_img = torch.from_numpy(tensor_img).unsqueeze(0).float()

        # If model is on CUDA, pass tensor to CUDA
        device = next(model.parameters()).device
        tensor_img = tensor_img.to(device)

        cam = GradCAM(model=model, target_layers=[target_layer])
        grayscale_cam = cam(input_tensor=tensor_img, targets=None)[0, :]
        visualization = show_cam_on_image(rgb_float, grayscale_cam, use_rgb=True)
        return visualization, None
    except Exception as e:
        return None, str(e)

# UNIFIED FOOD PREDICTOR CLASS
# ==============================================================================
class FoodPredictor:
    """
    Shared predictor supporting PyTorch (.pth) and ONNX (.onnx) backends.
    Includes temporal probability smoothing and safety rejection logic.
    """

    def __init__(
        self,
        backend: str = "torch",
        arch_name: str = "mobilenetv3",
        model_path: Optional[str] = None,
        threshold: float = 0.60,
        window_size: int = 10,
        device_str: Optional[str] = None,
    ):
        self.backend = backend.lower().strip()
        self.arch_name = arch_name.lower().strip()
        self.threshold = float(threshold)
        self.window_size = int(window_size)
        self.class_names = load_class_names()
        self.menu_df = load_menu()
        self.num_classes = len(self.class_names)

        # Sliding window deque for temporal smoothing across frames
        self.prob_history = deque(maxlen=self.window_size)

        # Device setup
        self.device_str = device_str
        self.torch_device = None
        self.model = None
        self.target_layer = None
        self.ort_session = None

        # Resolve model weights path
        self.model_path = self._resolve_model_path(model_path)

        # Initialize requested backend
        if self.backend == "torch":
            self._init_pytorch()
        elif self.backend == "onnx":
            self._init_onnx()
        else:
            raise ValueError(f"Invalid backend '{self.backend}'. Choose 'torch' or 'onnx'.")

    def _resolve_model_path(self, custom_path: Optional[str]) -> Path:
        """Finds model path or auto-detects from app/model/."""
        if custom_path:
            p = Path(custom_path)
            if p.exists():
                return p

        # Auto-detect default names
        if self.backend == "torch":
            preferred = MODEL_DIR / f"{self.arch_name}.pth"
            if preferred.exists():
                return preferred
            # Any .pth in MODEL_DIR
            alt = list(MODEL_DIR.glob("*.pth"))
            if alt:
                return alt[0]
            return preferred
        else:
            preferred = MODEL_DIR / "food_model.onnx"
            if preferred.exists():
                return preferred
            alt = list(MODEL_DIR.glob("*.onnx"))
            if alt:
                return alt[0]
            return preferred

    def _init_pytorch(self):
        """Loads PyTorch model on CUDA if available, else CPU."""
        import torch

        # Device selection: use GPU if available
        if self.device_str:
            self.torch_device = torch.device(self.device_str)
        else:
            self.torch_device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

        if not self.model_path.exists():
            raise FileNotFoundError(
                f"PyTorch weights not found at: '{self.model_path}'\n"
                f"Please download 'mobilenetv3.pth' from Google Drive into 'app/model/'."
            )

        model, target_layer = build_pytorch_architecture(self.arch_name, self.num_classes)
        state_dict = torch.load(self.model_path, map_location=self.torch_device)
        model.load_state_dict(state_dict)
        model.to(self.torch_device)
        model.eval()

        self.model = model
        self.target_layer = target_layer

    def _init_onnx(self):
        """Loads ONNX model using onnxruntime with CPU or CUDA provider."""
        try:
            import onnxruntime as ort
        except ImportError as e:
            raise ImportError(
                "onnxruntime is not installed. Please run: pip install onnxruntime"
            ) from e

        if not self.model_path.exists():
            raise FileNotFoundError(
                f"ONNX model file not found at: '{self.model_path}'\n"
                f"Please export or download 'food_model.onnx' into 'app/model/'."
            )

        # Use CUDAExecutionProvider if available, else CPUExecutionProvider
        available_providers = ort.get_available_providers()
        providers = []
        if "CUDAExecutionProvider" in available_providers and self.device_str != "cpu":
            providers.append("CUDAExecutionProvider")
        providers.append("CPUExecutionProvider")

        self.ort_session = ort.InferenceSession(str(self.model_path), providers=providers)

    def reset_smoothing(self):
        """Clears the temporal probability sliding window."""
        self.prob_history.clear()

    def _forward(self, input_tensor_np: np.ndarray) -> np.ndarray:
        """Executes forward pass through PyTorch or ONNX and returns 1D softmax array."""
        if self.backend == "torch":
            import torch
            tensor = torch.from_numpy(input_tensor_np).to(self.torch_device).float()
            with torch.no_grad():
                logits = self.model(tensor)
                probs = torch.softmax(logits, dim=1).cpu().numpy()[0]
            return probs
        else:
            # ONNX Runtime forward pass
            input_name = self.ort_session.get_inputs()[0].name
            raw_outputs = self.ort_session.run(None, {input_name: input_tensor_np})
            logits = raw_outputs[0][0]
            return softmax(logits)

    def predict_array(self, rgb_array: np.ndarray, smooth: bool = True) -> Dict[str, Any]:
        """
        Runs prediction on an RGB numpy array [height, width, 3].
        Applies center crop, ImageNet normalization, forward pass, smoothing, and thresholding.
        """
        # Preprocess to [1, 3, 224, 224]
        tensor_np = preprocess_numpy_image(rgb_array)

        # Forward pass -> 1D probabilities across classes
        raw_probs = self._forward(tensor_np)

        # Temporal smoothing over the last N frames
        if smooth:
            self.prob_history.append(raw_probs)
            smoothed_probs = np.mean(self.prob_history, axis=0)
        else:
            smoothed_probs = raw_probs

        # Top-1 index and confidence
        top_idx = int(np.argmax(smoothed_probs))
        top_conf = float(smoothed_probs[top_idx])
        top_class_raw = self.class_names[top_idx]

        # Safety & Not-Food Checks:
        # A prediction is valid food ONLY if class != 'not_food' AND confidence >= threshold
        is_not_food = (top_class_raw == "not_food")
        is_confident = (top_conf >= self.threshold)
        is_valid_food = (not is_not_food) and is_confident

        # Menu lookup for price and calories
        if is_valid_food:
            match_row = self.menu_df[self.menu_df["class"] == top_class_raw]
            if not match_row.empty:
                display_name = str(match_row.iloc[0]["display_name"])
                price = int(match_row.iloc[0]["price"])
                kcal = int(match_row.iloc[0]["kcal"])
            else:
                display_name = top_class_raw.replace("_", " ").title()
                price = 40
                kcal = 250
            status_text = display_name
        else:
            display_name = "No food / Unknown food"
            price = 0
            kcal = 0
            status_text = "No food / Unknown food"

        # Top 3 predictions for display
        top_k = min(3, len(self.class_names))
        top_indices = np.argsort(smoothed_probs)[::-1][:top_k]
        top_predictions = []
        for idx in top_indices:
            c_raw = self.class_names[idx]
            top_predictions.append({
                "class_raw": c_raw,
                "class_name": c_raw.replace("_", " ").title(),
                "confidence": float(smoothed_probs[idx])
            })

        return {
            "class_raw": top_class_raw,
            "class_name": display_name,
            "status_text": status_text,
            "confidence": top_conf,
            "is_food": is_valid_food,
            "is_not_food": is_not_food,
            "is_confident": is_confident,
            "price": price,
            "kcal": kcal,
            "all_probs": smoothed_probs,
            "top_predictions": top_predictions
        }

    def predict_cv2(self, frame_bgr: np.ndarray, smooth: bool = True) -> Dict[str, Any]:
        """Convenience method for OpenCV BGR camera frames."""
        import cv2
        rgb = cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB)
        return self.predict_array(rgb, smooth=smooth)

    def predict_pil(self, image_pil: Image.Image, smooth: bool = False) -> Dict[str, Any]:
        """Convenience method for PIL images (used by Streamlit upload/snapshot)."""
        rgb = np.array(image_pil.convert("RGB"))
        return self.predict_array(rgb, smooth=smooth)

# Converts own photos to clean JPGs, fixes folder names, and builds own_photos.zip
import os, zipfile
from pathlib import Path
from PIL import Image
import pillow_heif
pillow_heif.register_heif_opener()      # lets Pillow open AVIF/HEIC files

SRC = Path(r"C:\Projects\own_extracted")        # where you extracted the zip
DST = Path(r"C:\Projects\own_photos_clean")     # clean output folder
ZIP = Path(r"C:\Projects\own_photos.zip")       # final zip (outside OneDrive)

# Map any spelling of the folder name to the real class name
def norm(s): return s.lower().replace("_", "").replace(" ", "").replace("-", "")
CLASSES = {"vadapav": "vada_pav", "poha": "poha", "notfood": "not_food"}
EXT = {".jpg", ".jpeg", ".png", ".webp", ".avif", ".heic", ".bmp"}

DST.mkdir(parents=True, exist_ok=True)
counts = {}
for folder in SRC.rglob("*"):                     # search all sub-folders (handles extra parent folder)
    if folder.is_dir() and norm(folder.name) in CLASSES:
        cls = CLASSES[norm(folder.name)]
        out = DST / cls
        out.mkdir(exist_ok=True)
        n = counts.get(cls, 0)
        for f in sorted(folder.iterdir()):
            if f.suffix.lower() not in EXT:
                continue
            try:
                img = Image.open(f).convert("RGB")
                img.thumbnail((1024, 1024))        # limit size
                n += 1
                img.save(out / f"{cls}_{n:03d}.jpg", quality=90)
            except Exception as e:
                print("Skipped:", f.name, "->", e)
        counts[cls] = n

print("\nImages per class:", counts)
for c in CLASSES.values():
    if counts.get(c, 0) < 20:
        print(f"WARNING: {c} has fewer than 20 images!")

# Build the zip with class folders at the TOP level (forward slashes, Linux-safe)
with zipfile.ZipFile(ZIP, "w", zipfile.ZIP_DEFLATED) as z:
    for p in DST.rglob("*.jpg"):
        z.write(p, p.relative_to(DST).as_posix())
print("Created:", ZIP, "size MB:", round(ZIP.stat().st_size / 1e6, 1))
